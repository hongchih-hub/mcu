/* USER CODE BEGIN Header */
/**
  * @file           : main.c
  * @brief          : Main program body (Lab 3 Challenge)
  */
/* USER CODE END Header */

/* Includes ------------------------------------------------------------------*/
#include "main.h"

/* Private variables ---------------------------------------------------------*/
/* USER CODE BEGIN PV */

/* =========================
   State 定義 (不使用 enum)
   0 = NORMAL
   1 = WARNING
   2 = EMERGENCY (紅燈恆亮)
   ========================= */
#define STATE_NORMAL      0
#define STATE_WARNING     1
#define STATE_EMERGENCY   2

int state = STATE_NORMAL;

/* =========================
   Timing 變數 (使用 HAL_GetTick)
   ========================= */
uint32_t now = 0;
uint32_t previous = 0;
uint32_t warning_start = 0;

/* =========================
   STOP Event Flag (EXTI)
   ========================= */
volatile int STOP = 0;

/* =========================
   MODE Event & 非阻塞 Debounce
   ========================= */
int MODE_event = 0;
GPIO_PinState mode_reading = GPIO_PIN_SET;
GPIO_PinState mode_last_state = GPIO_PIN_SET;
GPIO_PinState mode_current_state = GPIO_PIN_SET;
uint32_t debounce_timer = 0;

/* USER CODE END PV */

/* Private function prototypes -----------------------------------------------*/
void SystemClock_Config(void);
static void MX_GPIO_Init(void);
void Error_Handler(void);

/**
  * @brief  程式進入點
  * @retval int
  */
int main(void)
{
  /* MCU 基本初始化 */
  HAL_Init();
  SystemClock_Config();
  MX_GPIO_Init();

  /* USER CODE BEGIN 2 */
  /* 開機進入 NORMAL 狀態 */
  previous = HAL_GetTick();
  HAL_GPIO_WritePin(GPIOC, GPIO_PIN_6, GPIO_PIN_SET);    // 綠燈 ON (PC6)
  HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_RESET);  // 黃燈 OFF (PC8)
  HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5, GPIO_PIN_RESET);  // 紅燈 OFF (PC5)
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    /* =====================================================
       Step 1：Read Events & Get Time
       ===================================================== */
    MODE_event = 0;
    now = HAL_GetTick();

    /* --- MODE 按鈕 (PC9) 非阻塞防彈跳偵測 --- */
    mode_reading = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_9);

    if (mode_reading != mode_last_state)
    {
        debounce_timer = now;
    }

    if ((now - debounce_timer) > 50)
    {
        if (mode_reading != mode_current_state)
        {
            mode_current_state = mode_reading;
            if (mode_current_state == GPIO_PIN_RESET)
            {
                MODE_event = 1;
            }
        }
    }
    mode_last_state = mode_reading;


    /* =====================================================
       Step 2：State Transition (狀態轉移)
       ===================================================== */

    // 1. 最高優先級：STOP 按下 -> 進入 EMERGENCY
    if (STOP == 1)
    {
        state = STATE_EMERGENCY;
        STOP = 0; // 清除 Flag

        // 進入 EMERGENCY 時：強制綠(PC6)、黃(PC8)全滅，紅(PC5)恆亮
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_6, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5, GPIO_PIN_SET);
    }
    // 2. NORMAL 狀態下按下 MODE -> 進入 WARNING
    else if ((state == STATE_NORMAL) && (MODE_event == 1))
    {
        state = STATE_WARNING;
        previous = now;
        warning_start = now;

        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_6, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_SET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5, GPIO_PIN_RESET);
    }
    // 3. WARNING 狀態下按下 MODE 或「逾時 5 秒」 -> 返回 NORMAL
    else if (state == STATE_WARNING)
    {
        if ((MODE_event == 1) || ((now - warning_start) >= 5000))
        {
            state = STATE_NORMAL;
            previous = now;

            HAL_GPIO_WritePin(GPIOC, GPIO_PIN_6, GPIO_PIN_SET);
            HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5, GPIO_PIN_RESET);
        }
    }

    /* =====================================================
       Step 3：State Behavior (持續行為與閃爍)
       ===================================================== */
    if (state == STATE_NORMAL)
    {
        if ((now - previous) >= 500) // 綠燈(PC6) 1Hz
        {
            previous = now;
            HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_6);
        }
    }
    else if (state == STATE_WARNING)
    {
        if ((now - previous) >= 250) // 黃燈(PC8) 2Hz
        {
            previous = now;
            HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_8);
        }
    }
    else if (state == STATE_EMERGENCY)
    {
        // 確保紅燈(PC5)恆亮，其餘熄滅
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_6, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5, GPIO_PIN_SET);
    }
  }
  /* USER CODE END WHILE */
}

/**
  * @brief 系統時脈設定
  */
void SystemClock_Config(void)
{----
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1_BOOST);

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSI;
  RCC_OscInitStruct.PLL.PLLM = RCC_PLLM_DIV4;
  RCC_OscInitStruct.PLL.PLLN = 85;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;
  RCC_OscInitStruct.PLL.PLLQ = RCC_PLLQ_DIV2;
  RCC_OscInitStruct.PLL.PLLR = RCC_PLLR_DIV2;
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK)
  {
    Error_Handler();
  }
}

/**
  * @brief GPIO 初始化
  */
static void MX_GPIO_Init(void)
{
  GPIO_InitTypeDef GPIO_InitStruct = {0};

  __HAL_RCC_GPIOC_CLK_ENABLE();
  __HAL_RCC_GPIOF_CLK_ENABLE();
  __HAL_RCC_GPIOA_CLK_ENABLE();
  __HAL_RCC_GPIOB_CLK_ENABLE();

  HAL_GPIO_WritePin(GPIOC, GPIO_PIN_5 | GPIO_PIN_6 | GPIO_PIN_8, GPIO_PIN_RESET);

  /* STOP 按鈕 (PC13) */
  GPIO_InitStruct.Pin = GPIO_PIN_13;
  GPIO_InitStruct.Mode = GPIO_MODE_IT_RISING;
  GPIO_InitStruct.Pull = GPIO_PULLDOWN;
  HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);

  /* 輸出腳位：PC5(紅), PC6(綠), PC8(黃) */
  GPIO_InitStruct.Pin = GPIO_PIN_5 | GPIO_PIN_6 | GPIO_PIN_8;
  GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
  GPIO_InitStruct.Pull = GPIO_NOPULL;
  GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
  HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);

  /* MODE 按鈕 (PC9) */
  GPIO_InitStruct.Pin = GPIO_PIN_9;
  GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
  GPIO_InitStruct.Pull = GPIO_PULLUP;
  HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);

  /* 啟用 PC13 外部中斷 */
  HAL_NVIC_SetPriority(EXTI15_10_IRQn, 1, 0);
  HAL_NVIC_EnableIRQ(EXTI15_10_IRQn);
}

/* USER CODE BEGIN 4 */

/* ========================================================
   這是最關鍵的一步！加入 IRQ Handler 避免程式跳入死迴圈
   ======================================================== */
void EXTI15_10_IRQHandler(void)
{
  HAL_GPIO_EXTI_IRQHandler(GPIO_PIN_13);
}

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
  if (GPIO_Pin == GPIO_PIN_13)
  {
    STOP = 1;
  }
}
/* USER CODE END 4 */

  * @brief 錯誤處理
  */
void Error_Handler(void)
{
  __disable_irq();
  while (1)
  {
  }
}
