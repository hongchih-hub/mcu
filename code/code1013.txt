/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  /* USER CODE BEGIN 1 */

  /* USER CODE END 1 */

  /* MCU Configuration--------------------------------------------------------*/

  /* Reset of all peripherals, Initializes the Flash interface and the Systick. */
  HAL_Init();

  /* USER CODE BEGIN Init */

  /* USER CODE END Init */

  /* Configure the system clock */
  SystemClock_Config();

  /* USER CODE BEGIN SysInit */

  /* USER CODE END SysInit */

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_TIM3_Init(); // CubeMX 產生的初始化，內部 PSC 依然保持課堂規定的 1699，ARR 保持 19999
  
  /* USER CODE BEGIN 2 */
  
  // 💡【破解法】: 在程式執行中，強制將 TIM3 的 ARR (AutoReload Register) 改為 999
  // 這樣 Timer 頻率 = 170MHz / (1699+1) = 100kHz。 100kHz / (999+1) = 100Hz (消除閃爍!)
  __HAL_TIM_SET_AUTORELOAD(&htim3, 999); 
  
  // 啟動 TIM3 Channel 3 (PC8) 的 PWM 硬體輸出
  HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_3); 
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    // 每 10 毫秒執行一次亮度調整 (非阻塞式，不會卡住 CPU)
    if (HAL_GetTick() - last_tick >= 10) 
    {
      last_tick = HAL_GetTick(); // 更新計時器起點

      if (is_increasing == 1) // 如果目前是漸亮狀態
      {
        pulse_value += 10; // 配合新的 ARR=999，每次增加 10。(100步 * 10ms = 1秒到達最亮)
        
        if (pulse_value >= 999) // 檢查是否達到最大亮度上限 (等於新的 ARR)
        {
          pulse_value = 999; // 限制在最大值
          is_increasing = 0; // 反轉狀態，改為漸暗 
        }
      }
      else // 如果目前是漸暗狀態
      {
        if (pulse_value <= 10) // 避免無號數減到變負數(溢位)
        {
          pulse_value = 0;     // 限制在最小值(完全熄滅)
          is_increasing = 1;   // 反轉狀態，改為漸亮
        }
        else 
        {
          pulse_value -= 10;   // 亮度減 10，慢慢變暗
        }
      }
      
      // 直接將計算好的 pulse_value 寫入硬體暫存器，即時改變 PWM 佔空比 (Duty Cycle)
      __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, pulse_value); 
    }
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
  }
  /* USER CODE END 3 */
}
